Table Especies {

  id_especie int [primary key]

  nombre_comun varchar

  nombre_cientifico varchar

  bioma varchar

  clima varchar

  habitos varchar

}


Table Recintos {

  id_recinto int [primary key]

  nombre_recinto varchar

  bioma varchar

  clima varchar

  capacidad_maxima int

  cupo_disponible int

  disponibilidad varchar

}


Table Animales {

  id_animal int [primary key]

  expediente varchar

  nombre_animal varchar

  sexo varchar

  fecha_ingreso date

  estado_salud varchar

  id_especie int

  id_recinto int

}


Table EvaluacionesMedicas {

  id_evaluacion int [primary key]

  id_animal int

  fecha_evaluacion date

  diagnostico varchar

  tratamiento varchar

  observaciones varchar

  estado_salud varchar

}


Table Movimientos {

  id_movimiento int [primary key]

  id_animal int

  fecha_movimiento date

  tipo_movimiento varchar

  recinto_origen int

  recinto_destino int

  destino_externo varchar

  motivo varchar

}
