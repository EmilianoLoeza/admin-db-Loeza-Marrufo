Table Producto {
  id_producto int [primary key]
  nombre_producto varchar
  precio_producto decimal
  stock_producto int
}

Table Compras {
  id_compra int [primary key]
  fecha_compra date
  total_compra decimal
  estado_compra varchar
}

Table DetalleCompra {
  id_detalle int [primary key]
  id_compra int
  id_producto int
  cantidad_producto int
  precio_unitario decimal
}

Table Pagos {
  id_pago int [primary key]
  id_compra int
  monto_pago decimal
  fecha_pago date
}

Ref compra_detalle: Compras.id_compra < DetalleCompra.id_compra
Ref producto_detalle: Producto.id_producto < DetalleCompra.id_producto
Ref compra_pago: Compras.id_compra < Pagos.id_compra
